import java.util.*;
public class Q3arrays
{
    int n;
    int arr[];
    Q3arrays(int a){
        n=a;
        arr=new int [n];
    }

    void input(){
        Scanner sc=new Scanner(System.in);
        for(int i=0;i<n;i++){
            arr[i]=sc.nextInt();
        }
    }

    void reverse(){
         int temp=0;
        
            for(int j=n-1,i=0;j>=n/2;j--,i++){
                arr[i]=arr[j];
                temp=arr[i];
                temp=arr[j];
            }
        }
    

    void Display(){
        for(int i=0;i<n;i++){
            System.out.println(arr[i]);
        }
    }

    public static void main(String args[]){
        Scanner sc=new Scanner(System.in);
        int n=sc.nextInt();
        Q3arrays obj=new Q3arrays(n);
        obj.input();
        obj.reverse();
        obj.Display();
    }
}
